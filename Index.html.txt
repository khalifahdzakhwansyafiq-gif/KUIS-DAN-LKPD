function toast(msg){
  var t = document.getElementById('toast');

function esc(s){ return String(s==null?'':s).replace(/&/g,'&amp;').replace(/</g,'&lt;')
  .replace(/>/g,'&gt;').replace(/"/g,'&quot;'); }
function uid(){ return Date.now().toString(36)+Math.random().toString(36).slice(2,8); }

function api(fn){
  var args = Array.prototype.slice.call(arguments, 1);
  return new Promise(function(res, rej){
    var call = google.script.run.withSuccessHandler(res).withFailureHandler(rej);
    call[fn].apply(call, args);
  });
}

function roomBadge(type, soft){
  var m = ROOM_META[type] || ROOM_META['latihan'];
  return '<span class="badge" style="background:'+(soft?m.soft:m.color)+
    ';color:'+(soft?m.color:'#fff')+'">'+m.label+'</span>';
}
function fmtDate(d){
  if(!d) return '';
  try{ return new Date(d+'T00:00').toLocaleDateString('id-ID',{day:'numeric',month:'short'}); }
  catch(e){ return d; }
}

function startQuiz(roomId){
  var room = S.rooms.find(function(r){ return String(r.id)===String(roomId); });
  if(!room) return;
  if(GROUP_TYPES.indexOf(room.type) >= 0){
    var g = prompt('Ruang ini dikerjakan berkelompok. Masukkan nama kelompokmu:', 'Kelompok 1');
    S.student.group = (g && g.trim()) ? g.trim() : 'Kelompok 1';
  } else S.student.group = '';
  S.activeRoom = room; S.answers = {}; S.qIndex = 0;
  // siapkan urutan acak kolom kanan untuk soal menjodohkan
  room.questions.forEach(function(q){
    if(q.type==='menjodohkan'){
      var idx = (q.pairs||[]).map(function(_,i){return i;});
      for(var i=idx.length-1;i>0;i--){ var j=Math.floor(Math.random()*(i+1)); var t=idx[i]; idx[i]=idx[j]; idx[j]=t; }
      q._shuffled = idx;
    }
  });
  S.view='quiz'; render();
}

/* ================= HALAMAN: Kuis ================= */
function vQuiz(){
  var room = S.activeRoom, qs = room.questions, q = qs[S.qIndex];
  var nav = qs.map(function(x,i){
    var a = S.answers[x.id];
    var done = a!==undefined && a!=='' && !(Array.isArray(a) && a.indexOf(-1)>=0);
    var cls = i===S.qIndex ? 'active' : (done?'done':'');
    return '<button class="'+cls+'" onclick="S.qIndex='+i+';render()">'+(i+1)+'</button>';
  }).join('');

  var body = '';
  if(q.type==='pg'){
    body = (q.options||[]).map(function(o,i){
      return '<button class="opt'+(S.answers[q.id]===i?' sel':'')+'" onclick="S.answers[\''+q.id+'\']='+i+';render()">'+
        '<span class="letter">'+String.fromCharCode(65+i)+'</span>'+esc(o)+'</button>';
    }).join('');
  } else if(q.type==='isian'){
    body = '<input class="input" placeholder="Tulis jawabanmu…" value="'+esc(S.answers[q.id]||'')+
      '" oninput="S.answers[\''+q.id+'\']=this.value">'+
      '<p class="small-note">Dinilai otomatis — perhatikan ejaan ya!</p>';
  } else if(q.type==='menjodohkan'){
    var cur = S.answers[q.id] || (q.pairs||[]).map(function(){return -1;});
    S.answers[q.id] = cur;
    body = (q.pairs||[]).map(function(p,i){
      var opts = '<option value="-1">Pilih pasangan…</option>' + q._shuffled.map(function(orig,di){
        return '<option value="'+di+'"'+(cur[i]===orig?' selected':'')+'>'+esc(q.pairs[orig].right)+'</option>';
      }).join('');
      return '<div class="pair-row"><div class="pair-left">'+esc(p.left)+'</div><span>↔</span>'+
        '<select class="input" onchange="S.answers[\''+q.id+'\']['+i+']='+ 
        'Number(this.value)>=0?S.activeRoom.questions[S.qIndex]._shuffled[Number(this.value)]:-1;render()">'+opts+'</select></div>';
    }).join('');
  } else {
    body = '<textarea class="input" rows="7" placeholder="Tuliskan jawabanmu selengkap mungkin…" '+
      'oninput="S.answers[\''+q.id+'\']=this.value">'+esc(S.answers[q.id]||'')+'</textarea>'+
      '<p class="small-note">Dinilai langsung oleh guru.</p>';
  }

  var meta = QT_META[q.type];
  app.innerHTML =
  '<div class="topbar"><span class="logo">'+esc(room.title)+'</span>'+roomBadge(room.type,true)+'</div>'+
  '<div class="qnav">'+nav+'</div>'+
  '<div class="card" style="margin-top:20px">'+
    '<span class="badge" style="border-color:var(--ink); background:'+meta.color+'22; color:'+meta.color+'">'+meta.label+'</span>'+
    '<span class="small-note" style="float:right">Soal '+(S.qIndex+1)+'/'+qs.length+' · '+q.points+' poin</span>'+
    '<h2 style="margin-top:14px; font-size:20px; line-height:1.4">'+esc(q.text)+'</h2>'+
    '<div style="margin-top:20px">'+body+'</div>'+
  '</div>'+
  '<div style="display:flex; justify-content:space-between; margin-top:18px">'+
    (S.qIndex>0 ? '<button class="btn" onclick="S.qIndex--;render()">← Sebelumnya</button>' : '<span></span>')+
    (S.qIndex<qs.length-1
      ? '<button class="btn primary" onclick="S.qIndex++;render()">Berikutnya →</button>'
      : '<button class="btn green" onclick="submitQuiz()">Kumpulkan Jawaban ✓</button>')+
  '</div>';
}
function submitQuiz(){
  if(!confirm('Yakin ingin mengumpulkan? Jawaban tidak bisa diubah setelah dikumpulkan.')) return;
  var room = S.activeRoom;
  toast('Mengumpulkan…');
  api('submitAnswers', {
    roomId: room.id, studentName: S.student.name, kelas: S.student.kelas,
    groupName: S.student.group || '', answers: S.answers
  }).then(function(res){
    S.lastResult = res; S.view='result'; render();
  }).catch(function(e){ toast('Gagal mengumpulkan: '+e.message); });
}

  S.editQuestion = {id:uid(), type:'pg', text:'', points:10, options:['','','',''], correctOption:0,

function renderQuestionModal(){
  var q = S.editQuestion;
  var el = document.getElementById('q-modal');
  var typeBtns = Object.keys(QT_META).map(function(t){
    return '<button class="btn small" style="'+(q.type===t?'background:'+QT_META[t].color+';color:#fff':'border-color:var(--line);color:var(--muted);box-shadow:none')+
      '" onclick="S.editQuestion.type=\''+t+'\';renderQuestionModal()">'+QT_META[t].label+'</button>';
  }).join(' ');

  var extra = '';
  if(q.type==='pg'){
    extra = '<label class="lbl">Pilihan & kunci (angka kunci = nomor pilihan, mulai 0)</label>'+
      q.options.map(function(o,i){
        return '<div style="display:flex; gap:8px; margin-bottom:8px; align-items:center">'+
          '<button class="btn small" style="'+(q.correctOption===i?'background:var(--green);color:#fff':'border-color:var(--line);box-shadow:none')+
          '" onclick="S.editQuestion.correctOption='+i+';renderQuestionModal()">'+String.fromCharCode(65+i)+'</button>'+
          '<input class="input" value="'+esc(o)+'" oninput="S.editQuestion.options['+i+']=this.value"></div>';
      }).join('');
  } else if(q.type==='isian'){
    extra = '<label class="lbl">Jawaban yang diterima (satu per baris)</label>'+

  } else if(q.type==='menjodohkan'){
    extra = '<label class="lbl">Pasangan kiri — kanan (satu baris = satu pasangan, pisahkan dengan tanda |)</label>'+
      '<textarea class="input" rows="4" placeholder="1/2 | 0,5&#10;1/4 | 0,25" oninput="S.editQuestion.pairs=this.value.split(\'\\n\').filter(function(l){return l.indexOf(\'|\')>=0}).map(function(l){var p=l.split(\'|\');return {left:p[0].trim(),right:p.slice(1).join(\'|\').trim()}})">'+
      esc((q.pairs||[]).map(function(p){return p.left+' | '+p.right;}).join('\n'))+'</textarea>'+
      '<p class="small-note">Kolom kanan diacak otomatis saat siswa mengerjakan.</p>';
  } else {
    extra = '<label class="lbl">Kata kunci rubrik (pisahkan koma, opsional)</label>'+
      '<input class="input" value="'+esc((q.keywords||[]).join(', '))+'" oninput="S.editQuestion.keywords=this.value.split(\',\').map(function(k){return k.trim()}).filter(Boolean)">';
  }

  '<div class="modal"><div>'+
    '<div style="display:flex; justify-content:space-between; align-items:center">'+
      '<h3>'+(S.editQuestionIndex>=0?'Ubah Soal':'Tambah Soal')+'</h3>'+
      '<button class="btn ghost small" onclick="document.getElementById(\'q-modal\').innerHTML=\'\'">Tutup</button></div>'+

    '<button class="btn primary" style="width:100%; justify-content:center; margin-top:18px" onclick="saveQuestion()">Simpan Soal</button>'+
  '</div></div>';
}
function saveQuestion(){
  var q = S.editQuestion;
  if(q.text.trim().length < 3){ toast('Teks soal wajib diisi!'); return; }
  if(q.type==='pg' && q.options.filter(function(o){return o.trim();}).length < 2){ toast('Minimal 2 pilihan!'); return; }
  if(q.type==='isian' && !(q.answers||[]).some(function(a){return a.trim();})){ toast('Isi jawaban yang diterima!'); return; }
  if(q.type==='menjodohkan' && (q.pairs||[]).length < 2){ toast('Minimal 2 pasangan!'); return; }
  if(q.type==='pg'){ // rapikan pilihan kosong & kunci
    var filled = q.options.map(function(o,i){return {o:o,i:i};}).filter(function(x){return x.o.trim();});
    var key = filled.findIndex(function(x){return x.i===q.correctOption;});
    q.options = filled.map(function(x){return x.o.trim();});
    q.correctOption = key>=0?key:0;
  }
  if(S.editQuestionIndex>=0) S.editRoom.questions[S.editQuestionIndex]=q;
  else S.editRoom.questions.push(q);
  render();
  toast('Soal ditambahkan ✓');
}

function gradeSub(subId){
  var s = S.subs.find(function(x){ return String(x.id)===String(subId); });
  var room = S.rooms.find(function(r){ return String(r.id)===String(s.roomId); });
  if(!room){ toast('Ruang sudah dihapus'); return; }
  var html = room.questions.map(function(q,i){
    var ans = s.answers[q.id];
    var ansTxt = '';
    if(q.type==='pg') ansTxt = ans!==undefined ? String.fromCharCode(65+Number(ans))+'. '+(q.options||[])[ans] : '—';
    else if(q.type==='menjodohkan') ansTxt = (q.pairs||[]).map(function(p,j){
      return esc(p.left)+' → '+esc((q.pairs||[])[ans[j]] ? q.pairs[ans[j]].right : '—'); }).join('<br>');
    else ansTxt = esc(ans||'—');
    var out = '<div style="border:2px solid var(--line); border-radius:14px; padding:12px; margin-bottom:10px">'+
      '<p class="small-note">Soal '+(i+1)+' · '+q.points+' poin</p>'+
      '<p style="font-weight:700; font-size:14px">'+esc(q.text)+'</p>'+
      '<p style="margin-top:6px; font-size:14px">'+ansTxt+'</p>';
    if(q.type==='essay'){
      var cur = (s.essayScores||{})[q.id];
      out += '<div style="margin-top:10px; background:#ffe3f2; border-radius:12px; padding:10px">'+
        '<label class="lbl" style="margin:0; color:var(--pink)">Nilai essay (maks '+q.points+')</label>'+
        '<input type="number" min="0" max="'+q.points+'" class="input" style="width:100px; margin-top:6px" id="es-'+q.id+'" value="'+(cur!==undefined?cur:'')+'">'+
        ((q.keywords||[]).length?'<p class="small-note">Acuan: '+esc(q.keywords.join(' · '))+'</p>':'')+'</div>';
    }
    return out+'</div>';
  }).join('');
  document.getElementById('grade-modal').innerHTML =
  '<div class="modal"><div>'+
    '<div style="display:flex; justify-content:space-between; align-items:center">'+
      '<h3>'+esc(s.studentName)+(s.groupName?' · '+esc(s.groupName):'')+'</h3>'+
      '<button class="btn ghost small" onclick="document.getElementById(\'grade-modal\').innerHTML=\'\'">Tutup</button></div>'+
    '<div style="margin-top:14px">'+html+'</div>'+
    '<button class="btn primary" style="width:100%; justify-content:center" onclick=\'saveGrades('+JSON.stringify(s.id).replace(/"/g,'&quot;')+')\'>Simpan Nilai Essay</button>'+
  '</div></div>';
  window._gradingSub = s;
}
function saveGrades(subId){
  var s = S.subs.find(function(x){ return String(x.id)===String(subId); });
  var room = S.rooms.find(function(r){ return String(r.id)===String(s.roomId); });
  var scores = s.essayScores || {};
  var essayIds = [];
  room.questions.forEach(function(q){
    if(q.type!=='essay') return;
    essayIds.push(q.id);
    var inp = document.getElementById('es-'+q.id);
    if(inp && inp.value!=='') scores[q.id] = Math.min(q.points, Math.max(0, Number(inp.value)||0));
  });
  api('saveEssayScores', S.pin, subId, scores, essayIds)
    .then(function(){ return api('getSubmissions', S.pin); })
    .then(function(subs){ S.subs = subs; render(); toast('Nilai tersimpan ✓'); })
    .catch(function(e){ toast('Gagal: '+e.message); });
}
